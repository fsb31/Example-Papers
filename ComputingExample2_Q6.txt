import numpy as np
import matplotlib.pyplot as plt
import soundfile as sf



def DFT(data): #very slow
    N = len(data)
    X = np.zeros(N, dtype=complex)
    for k in range(0, N-1):
        for n in range(0, N-1):
            X[k] += data[n] * complex(np.cos(-2 * np.pi * k * n/N), np.sin(-2 * np.pi * k * n/N))
    return X


def PadData(data):
    N = len(data)
    if (N <= 1): return data
    if (N.bit_count() != 1): #check if not a power of 2 (if it is, only 1 bit will be set)
        New = int(np.pow(2, np.ceil(np.log2(N)))) - N #get number of elements that need to be added to array to make length a power of 2
        return np.pad(data, (0, New)) #pad with zeros


def FFT(data):
    N = len(data)

    if (N <= 1): return data
    
    E = FFT(data[::2]) #split data into even and odd elements, and recurse
    O = FFT(data[1::2])

    conv = np.exp(-2j*np.pi*np.arange(N) * 1/N) #the common multiplication factor 

    return np.concatenate([E + O*conv[:N//2], E + O*conv[N//2:]]) #add all the elements back together



#read data
data, rate = sf.read("Armstrong_Small_Step.ogg")
no_samples = len(data)
T = no_samples // rate #Sample length
print(f"Audio Length: {T}s, Sample Rate: {rate}.") 


t = np.linspace(0, T, no_samples, endpoint=False) #Sample times

#tmp_data = np.sin((2*np.pi*240) * t) + np.sin((2*np.pi *440) * t) #temporary data

#fft needs data length to be a power of 2 for recursion
data = PadData(data)

#fft
X = FFT(data)

#Remove 'mirrored' data
N = len(X)//2
X = X[:N]/N



#Plot
freq_axis = np.linspace(0, rate/2, N)

#plt.plot(t, tmp_data)
plt.plot(freq_axis, np.abs(X))
plt.xlabel("Freq (Hz)")
plt.ylabel("Amplitude")
plt.show()
